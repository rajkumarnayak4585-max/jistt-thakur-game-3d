function createRoom(){ if(peer) peer.destroy(); host=true; myIndex=0; connections=[]; const shortId=String(Math.floor(100000+Math.random()*900000)); say("Creating short Room ID..."); const shortId = String(Math.floor(100000 + Math.random() * 900000)); peer = new Peer(shortId); peer.on("open",id=>{ showRoom(id); say("Room ready! Share this 6-digit ID with friends."); freshGame(); startHost(); }); peer.on("connection",conn=>{ conn.on("open",()=>{ let slot=[1,2,3].find(i=>!connections.some(c=>c.slot===i)); if(slot===undefined){ conn.send({type:"full"}); conn.close(); return; } conn.slot=slot; connections.push(conn); conn.send({type:"welcome",slot,players,fruit}); broadcast(); conn.on("data",data=>{ if(data.type==="dir" && players[slot]){ const p=players[slot]; if(!(data.dir.x===-p.dir.x && data.dir.y===-p.dir.y)){ p.next=data.dir; } } }); conn.on("close",()=>{ connections=connections.filter(c=>c!==conn); if(players[slot]) players[slot].alive=false; broadcast(); }); }); }); peer.on("error",e=>{ say("Room error: "+e.type+". Try creating another room."); }); } function joinRoom(){ const id=document.getElementById("roomInput").value.trim(); if(!/^\d{6}$/.test(id)){ say("Please enter a valid 6-digit Room ID."); return; } if(peer) peer.destroy(); host=false; myIndex=-1; say("Joining room..."); peer=new Peer(); peer.on("open",()=>{ hostConn=peer.connect(id,{reliable:true}); hostConn.on("open",()=>{ showRoom(id); say("Connected! Waiting for game..."); }); hostConn.on("data",data=>{ if(data.type==="full"){ say("Room is full!"); return; } if(data.type==="welcome"){ myIndex=data.slot; players=data.players; fruit=data.fruit; say("Joined as "+names[myIndex]+"!"); draw(); } if(data.type==="state"){ players=data.players; fruit=data.fruit; draw(); } }); hostConn.on("close",()=>say("Host disconnected.")); }); peer.on("error",e=>say("Join error: "+e.type)); }
